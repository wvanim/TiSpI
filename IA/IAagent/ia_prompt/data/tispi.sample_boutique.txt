P moduleRequestPhp
  doc "Generic PHP request module, isolated timeline."
  tags "REQUEST_PENDING, REQUEST_OK, REQUEST_ERROR"
  contract(
    timeline_isolated = true,
    requires: Tispi.phpRequest(url, params),
    emit: REQUEST_PENDING, REQUEST_OK, REQUEST_ERROR,
    emit_data: --message
  )
  params(
    string --url,
    string --params,
    P --sender,
    func --on-success,
    func --on-error
  )
  modele (
    PF isolation : group
      P request
        T tag[1,REQUEST_PENDING ; 2,REQUEST_OK ; 3,REQUEST_ERROR]
        T stop[0]
        T face[1,fPending ; 2,fOk ; 3,fError]
        T action_immediate[
            0: Tispi.phpRequest(--url, --params) ;
            2: send(--sender, --on-success, --message) ;
            3: send(--sender, --on-error, --message)
        ]
        F fPending text "Requête en cours…"
        F fOk      text ""
        F fError   text ""
  )
